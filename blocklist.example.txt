# Cerberus policy file
#
# Directives:
#
#   mode monitor     report matching access only (default)
#   mode enforce     deny matching access with -EPERM
#
#   name <file>      match a file name anywhere on the filesystem
#   path <path>      match the parent-directory/file-name pair
#                    (the last two components of the path)
#
#   proc <comm> <target>     deny one process access to target
#   uid <uid> <target>       deny one uid access to target
#
#   sandbox <comm>           allowlist-only mode for a process;
#                            an empty allowlist denies all opens
#   allow <comm> <target>    allow a sandboxed process a target
#
#   net-mode monitor|enforce network policy mode (separate
#                            from the file-access mode)
#   net-block <ip>           block outgoing connections to an
#                            IPv4 or IPv6 address
#   net-allow <ip>           explicitly allow connections to an
#                            IPv4 or IPv6 address
#
# IP policy precedence: block rule > allow rule > default
# (allowed). Unlisted addresses are never denied.
#
# <target> is either a plain file name or a path containing '/'.
#
# Lines starting with '#' are ignored.

mode monitor

# Example name rule: any file called sensitive.txt
# name sensitive.txt

# Example path rule: a sensitive.txt inside a directory
# named secrets, anywhere on the filesystem
# path /var/secrets/sensitive.txt

# Example process rule: bash may not read the shadow file
# proc bash /etc/shadow

# Example uid rule: uid 1000 may not read the shadow file
# uid 1000 /etc/shadow

# Example sandbox: the "backup" tool may only read its config.
# Static binaries work best, because a strict sandbox also
# blocks the dynamic loader's file opens.
# sandbox backup
# allow backup /etc/backup.conf

# Example network rules: block outgoing connections to
# these addresses (net-mode decides report vs deny).
# net-mode enforce
# net-block 192.0.2.10
# net-block 2001:db8::1
