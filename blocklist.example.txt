# Cerberus policy file
#
# Directives:
#
#   mode monitor    report matching access only (default)
#   mode enforce    deny matching access with -EPERM
#
#   name <file>     match a file name anywhere on the filesystem
#   path <path>     match the parent-directory/file-name pair
#                   (the last two components of the path)
#
#   proc <comm> <target>     deny one process access to target
#   uid <uid> <target>       deny one uid access to target
#
#   sandbox <comm>           allowlist-only mode for a process;
#                            an empty allowlist denies all opens
#   allow <comm> <target>    allow a sandboxed process a target
#
# <target> is either a plain file name or a path containing '/'.
#
# Lines starting with '#' are ignored.

mode monitor

# Example name rule: any file called sensitive.txt
# name sensitive.txt

# Example path rule: a sensitive.txt inside a directory
# named secrets, anywhere on the filesystem
# path /var/secrets/sensitive.txt

# Example process rule: bash may not read the shadow file
# proc bash /etc/shadow

# Example uid rule: uid 1000 may not read the shadow file
# uid 1000 /etc/shadow

# Example sandbox: the "backup" tool may only read its config.
# Static binaries work best, because a strict sandbox also
# blocks the dynamic loader's file opens.
# sandbox backup
# allow backup /etc/backup.conf

# ----------------------------------------------------------------
# Network policy (Phase 5)
#
#   net-mode monitor | enforce | deny-all
#   block-domain <domain>     deny domain (+ subdomains), enforced
#                             on addresses observed in plaintext
#                             DNS answers
#   allow-domain <domain>     permit domain (+ subdomains), honored
#                             under deny-all
#   block-ip <ip>             deny one address (IPv4 or IPv6)
#   allow-ip <ip>             permit one address under deny-all
#   block-suspicious on|off   kernel-side denial for watched shell
#                             processes to unmapped destinations
#                             (default off; alerts are report-only)
#   net-dns-port on|off       port-53 exemption in deny-all
#                             (default on)
#   add-shellwatch <comm>     extend the reverse-shell watch list
#
# DNS-based domain rules are only as complete as plaintext
# UDP-53 visibility: DoH, DoT and resolver caches are not
# mirrored. Direct-IP connections carry no domain context and
# are logged, not guessed. Shared hosting IPs are flagged in
# the alerts instead of being silently collateralized.
#
# deny-all via the config file is explicit human intent and
# takes effect directly with a printed warning; from the
# interactive CLI it requires a confirmation step.
# ----------------------------------------------------------------

net-mode monitor

# block-domain evil.example.test
# allow-domain tools.example.test
# block-ip 203.0.113.66
# allow-ip 198.51.100.3
# add-shellwatch nc
# block-suspicious off
